# TTS dependencies - require Python 3.9-3.11
# These are installed in venv311 (Python 3.11) by the Makefile setup target.

# Text-to-Speech Engine
# NOTE: TTS 0.21.0 and 0.22.0 have compatibility issues with transformers 5.1.0
# (missing BeamSearchScorer). If you encounter import errors, you may need to
# use an older TTS version or wait for a fix.

# Transformers - pinned to 4.36.0 for BeamSearchScorer compatibility with TTS
# TTS requires BeamSearchScorer which was removed in transformers 5.x
transformers==4.36.0

# NumPy - pinned for compatibility with numba and gruut
# numba requires numpy<2.4, gruut requires numpy<2.0.0
numpy>=1.19.0,<2.0.0
